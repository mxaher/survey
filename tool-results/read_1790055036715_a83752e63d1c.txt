     1→"use client";
     2→
     3→/**
     4→ * 4-step employee survey wizard (Task 3-a, spec §8.2).
     5→ *
     6→ * Steps:
     7→ *   1) بيئة العمل — environment question snapshots (single-select scale)
     8→ *   2) تقييم القيادات — pick an executive → leadership snapshots (5(+1) scale)
     9→ *   3) البيئة المستقبلية — future snapshots (single_choice RadioGroup /
    10→ *      multi_choice Checkbox with maxSelections)
    11→ *   4) المراجعة والإرسال — completion summary + confirm checkbox → success
    12→ *
    13→ * Server-state queries:
    14→ *   - participation-status: tells us which sections are already submitted
    15→ *     (drives step skipping + the "تم تقديم … مسبقًا" banners).
    16→ *   - executives: list of available executives + already-evaluated IDs.
    17→ *   - executive questions: leadership snapshots for the selected exec.
    18→ *
    19→ * Mutations:
    20→ *   - submitEnvironment, submitFuture, submitEvaluation — each refetches
    21→ *     participation-status + executives on success so the wizard stays in
    22→ *     sync with the server's view of the employee's progress.
    23→ *
    24→ * In-progress answer state lives in `useWizardStore` (Zustand, persisted to
    25→ * sessionStorage) — see `wizard-store.ts`. Each executive eval uses local
    26→ * React state (transient; resets when the user picks a different exec).
    27→ *
    28→ * Resilience: on mount we derive the starting step from participation-status,
    29→ * so a refresh after submitting environment restarts the user at step 2
    30→ * automatically (the wizard survives a refresh without needing client state
    31→ * for the step itself).
    32→ */
    33→import { useEffect, useMemo, useState } from "react";
    34→import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
    35→import { motion, AnimatePresence } from "framer-motion";
    36→import {
    37→  ArrowRight,
    38→  ArrowLeft,
    39→  Send,
    40→  CheckCircle2,
    41→  AlertCircle,
    42→  RefreshCw,
    43→  ChevronLeft,
    44→  ShieldCheck,
    45→  Lock,
    46→} from "lucide-react";
    47→import { Button } from "@/components/ui/button";
    48→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    49→import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";
    50→import { Progress } from "@/components/ui/progress";
    51→import { Badge } from "@/components/ui/badge";
    52→import { Checkbox } from "@/components/ui/checkbox";
    53→import { Label } from "@/components/ui/label";
    54→import { Skeleton } from "@/components/ui/skeleton";
    55→import { useToast } from "@/hooks/use-toast";
    56→import { MESSAGES } from "@/lib/messages";
    57→import { QuestionCard } from "./question-card";

    62→  ActiveCampaign,
    63→  ExecutivesResponse,
    64→  ParticipationStatus,
    65→  ExecutiveQuestionsResponse,
    66→} from "./types";
    67→
    68→const STEPS = [
    69→  { key: "environment", labelAr: "بيئة العمل" },
    70→  { key: "leadership", labelAr: "تقييم القيادات" },
    71→  { key: "future", labelAr: "البيئة المستقبلية" },
    72→  { key: "review", labelAr: "المراجعة والإرسال" },
    73→] as const;
    74→
    75→interface SurveyWizardProps {
    76→  campaign: ActiveCampaign;
    77→  /** Called when the user clicks the final submit on step 4. */
    78→  onFinish: () => void;
    79→  /** Called when the user explicitly aborts the wizard (cancel). */
    80→  onCancel: () => void;
    81→}
    82→
    83→export function SurveyWizard({ campaign, onFinish, onCancel }: SurveyWizardProps) {
    84→  const queryClient = useQueryClient();
    85→  const { toast } = useToast();
    86→
    87→  // ─── Server-state queries ─────────────────────────────────────────────
    88→  const participationStatusQuery = useQuery<ParticipationStatus>({
    89→    queryKey: ["employee-participation-status", campaign.id],
    90→    queryFn: () =>
    91→      fetchEmployeeApi<ParticipationStatus>(
    92→        `/api/employee/participation-status?campaignId=${encodeURIComponent(campaign.id)}`
    93→      ),
    94→    enabled: !!campaign.id,
    95→    retry: false,
    96→  });
    97→
    98→  const executivesQuery = useQuery<ExecutivesResponse>({
    99→    queryKey: ["employee-executives", campaign.id],
   100→    queryFn: () =>
   101→      fetchEmployeeApi<ExecutivesResponse>(
   102→        `/api/employee/executives?campaignId=${encodeURIComponent(campaign.id)}`
   103→      ),
   104→    enabled: !!campaign.id,
   105→    retry: false,
   106→  });
   107→
   108→  // ─── Wizard store (client-side in-progress answer state) ──────────────
   109→  const envAnswers = useWizardStore((s) => s.envAnswers);
   110→  const futureAnswers = useWizardStore((s) => s.futureAnswers);
   111→  const currentExecutiveId = useWizardStore((s) => s.currentExecutiveId);
   112→  const confirmedFinal = useWizardStore((s) => s.confirmedFinal);
   113→  const setEnvAnswer = useWizardStore((s) => s.setEnvAnswer);
   114→  const setFutureAnswer = useWizardStore((s) => s.setFutureAnswer);
   115→  const setCurrentExecutive = useWizardStore((s) => s.setCurrentExecutive);
   116→  const setConfirmedFinal = useWizardStore((s) => s.setConfirmedFinal);
   117→  const resetWizard = useWizardStore((s) => s.reset);
   118→
   119→  // ─── Local UI state ──────────────────────────────────────────────────
   120→  const [step, setStep] = useState<number>(1);
   121→  const [initialized, setInitialized] = useState(false);
   122→
   123→  // Local answer state for the current executive evaluation (transient —
   124→  // reset whenever the user picks a different executive).
   125→  const [executiveAnswers, setExecutiveAnswers] = useState<
   126→    Record<string, string>
   127→  >({});
   128→  // Track the executive id the local answers belong to (so we can reset
   129→  // when the user picks a different one).
   130→  const [answersForExecId, setAnswersForExecId] = useState<string | null>(null);
   131→
   132→  // ─── Derive starting step from server-side participation-status ───────
   133→  // React-idiomatic render-phase state adjustment: compute the "floor" step
   134→  // from server truth (so a refresh after submitting env lands the user at
   135→  // step 2), then advance the local `step` only on first load (before the
   136→  // user has moved manually). After initialization, the user's own next/back
   137→  // clicks drive `step`; the derived floor only re-applies if the server's
   138→  // truth changes (e.g., a tab-submitted section is reflected via refetch).
   139→  const ps = participationStatusQuery.data;
   140→  const derivedStep = useMemo(() => {
   141→    if (!ps) return 1;
   142→    let s = 1;
   143→    const envDone =
   144→      !campaign.enableEnvironmentSurvey || ps.environmentSubmitted;
   145→    if (envDone) {
   146→      const evaluated = new Set(ps.evaluatedExecutiveIds);
   147→      const totalAssigned = executivesQuery.data?.executives.length ?? 0;
   148→      const allExecsDone =
   149→        totalAssigned > 0 && evaluated.size >= totalAssigned;
   150→      const singleEvalDone =
   151→        !campaign.allowMultipleExecutiveEvaluations && evaluated.size >= 1;
   152→      const step2Done = allExecsDone || singleEvalDone;
   153→      if (step2Done) {
   154→        const futureDone =
   155→          !campaign.enableFutureSurvey || ps.futureSubmitted;
   156→        s = futureDone ? 4 : 3;
   157→      } else {
   158→        s = 2;
   159→      }
   160→    }
   161→    return s;
   162→  }, [ps, campaign, executivesQuery.data]);
   163→
   164→  // Apply the derived step once on first load (initialized=false → true).
   165→  // After that, only advance (never move backwards) if the server's truth
   166→  // changes — e.g., user submits future from another tab.
   167→  if (ps && !initialized) {
   168→    setInitialized(true);
   169→    setStep(derivedStep);
   170→  } else if (ps && initialized && derivedStep > step) {
   171→    // Server says we should be further along than we are — bump up.
   172→    setStep(derivedStep);
   173→  }
   174→
   175→  // ─── beforeunload warning (in-progress answers) ──────────────────────
   176→  useEffect(() => {
   177→    const hasInProgressAnswers =
   178→      Object.keys(envAnswers).length > 0 ||
   179→      Object.keys(futureAnswers).length > 0 ||
   180→      Object.keys(executiveAnswers).length > 0;
   181→    if (!hasInProgressAnswers) return;
   182→    const handler = (e: BeforeUnloadEvent) => {
   183→      e.preventDefault();
   184→      e.returnValue =
   185→        "هل تريد المغادرة؟ ستفقد تقدمك غير المُرسل.";
   186→      return e.returnValue;
   187→    };
   188→    window.addEventListener("beforeunload", handler);
   189→    return () => window.removeEventListener("beforeunload", handler);
   190→  }, [envAnswers, futureAnswers, executiveAnswers]);
   191→
   192→  // ─── Refetch helpers (post-submit invalidation) ──────────────────────
   193→  const invalidateAll = () => {
   194→    queryClient.invalidateQueries({
   195→      queryKey: ["employee-participation-status", campaign.id],
   196→    });
   197→    queryClient.invalidateQueries({
   198→      queryKey: ["employee-executives", campaign.id],
   199→    });
   200→  };
   201→
   202→  // ─── Mutations ───────────────────────────────────────────────────────
   203→  const envMutation = useMutation({
   204→    mutationFn: async () => {
   205→      const answers = Object.entries(envAnswers).map(
   206→        ([questionSnapshotId, selectedValue]) => ({
   207→          questionSnapshotId,
   208→          selectedValue,
   209→        })
   210→      );
   211→      return fetchEmployeeApi<{ message: string }>(
   212→        "/api/employee/environment/submit",
   213→        {
   214→          method: "POST",
   215→          headers: { "Content-Type": "application/json" },
   216→          body: JSON.stringify({ campaignId: campaign.id, answers }),
   217→        }
   218→      );
   219→    },
   220→    onSuccess: (data) => {
   221→      toast({
   222→        title: "تم بنجاح",
   223→        description: data?.message ?? MESSAGES.submissionSuccess,
   224→      });
   225→      invalidateAll();
   226→      setStep(2);
   227→    },
   228→    onError: (err: Error) => {
   229→      const status = err instanceof ApiError ? err.status : 0;
   230→      toast({
   231→        variant: "destructive",
   232→        title: "تعذّر الإرسال",
   233→        description: err.message,
   234→      });
   235→      // 409 → already submitted; refetch so the banner shows.
   236→      if (status === 409) {
   237→        invalidateAll();
   238→        setStep(2);
   239→      }
   240→    },
   241→  });
   242→
   243→  const futureMutation = useMutation({
   244→    mutationFn: async () => {
   245→      const answers = Object.entries(futureAnswers).map(
   246→        ([questionSnapshotId, selectedValues]) => ({
   247→          questionSnapshotId,
   248→          selectedValues,
   249→        })
   250→      );
   251→      return fetchEmployeeApi<{ message: string }>(
   252→        "/api/employee/future/submit",
   253→        {
   254→          method: "POST",
   255→          headers: { "Content-Type": "application/json" },
   256→          body: JSON.stringify({ campaignId: campaign.id, answers }),
   257→        }
   258→      );
   259→    },
   260→    onSuccess: (data) => {
   261→      toast({
   262→        title: "تم بنجاح",
   263→        description: data?.message ?? MESSAGES.submissionSuccess,
   264→      });
   265→      invalidateAll();
   266→      setStep(4);
   267→    },
   268→    onError: (err: Error) => {
   269→      const status = err instanceof ApiError ? err.status : 0;
   270→      toast({
   271→        variant: "destructive",
   272→        title: "تعذّر الإرسال",
   273→        description: err.message,
   274→      });
   275→      if (status === 409) {
   276→        invalidateAll();
   277→        setStep(4);
   278→      }
   279→    },
   280→  });
   281→
   282→  const evalMutation = useMutation({
   283→    mutationFn: async ({
   284→      executiveId,
   285→      answers,
   286→    }: {
   287→      executiveId: string;
   288→      answers: Record<string, string>;
   289→    }) => {
   290→      const payload = Object.entries(answers).map(
   291→        ([questionSnapshotId, selectedValue]) => ({
   292→          questionSnapshotId,
   293→          selectedValue,
   294→        })
   295→      );
   296→      return fetchEmployeeApi<{ message: string }>(
   297→        "/api/employee/evaluations/submit",
   298→        {
   299→          method: "POST",
   300→          headers: { "Content-Type": "application/json" },
   301→          body: JSON.stringify({
   302→            campaignId: campaign.id,
   303→            executiveId,
   304→            answers: payload,
   305→          }),
   306→        }
   307→      );
   308→    },
   309→    onSuccess: (data) => {
   310→      toast({
   311→        title: "تم بنجاح",
   312→        description: data?.message ?? MESSAGES.submissionSuccess,
   313→      });
   314→      // Reset local exec answers + clear selected exec so the picker
   315→      // refetches and the dropdown shows the remaining executives.
   316→      setExecutiveAnswers({});
   317→      setAnswersForExecId(null);
   318→      setCurrentExecutive(null);
   319→      invalidateAll();
   320→    },
   321→    onError: (err: Error) => {
   322→      const status = err instanceof ApiError ? err.status : 0;
   323→      toast({
   324→        variant: "destructive",
   325→        title: "تعذّر الإرسال",
   326→        description: err.message,
   327→      });
   328→      if (status === 409) {
   329→        // Already evaluated this exec → reset selection + refetch.
   330→        setExecutiveAnswers({});
   331→        setAnswersForExecId(null);
   332→        setCurrentExecutive(null);
   333→        invalidateAll();
   334→      }
   335→    },
   336→  });
   337→
   338→  // ─── Executive questions query (only when an exec is selected) ────────
   339→  const execQuestionsQuery = useQuery<ExecutiveQuestionsResponse>({
   340→    queryKey: ["employee-executive-questions", campaign.id, currentExecutiveId],
   341→    queryFn: () =>
   342→      fetchEmployeeApi<ExecutiveQuestionsResponse>(
   343→        `/api/employee/executives/${encodeURIComponent(currentExecutiveId ?? "")}/questions`
   344→      ),
   345→    enabled: !!currentExecutiveId,
   346→    retry: false,
   347→  });
   348→
   349→  // Reset local exec answers when the user picks a different exec. We do this
   350→  // via a render-phase state adjustment (the React-idiomatic pattern for
   351→  // "adjust state when a prop changes") so we avoid the
   352→  // `setState-in-effect` lint warning and the cascading render that comes
   353→  // with it.
   354→  if (currentExecutiveId !== answersForExecId) {
   355→    setExecutiveAnswers({});
   356→    setAnswersForExecId(currentExecutiveId);
   357→  }
   358→
   359→  // ─── Derived state for navigation button enablement ──────────────────
   360→  const executives = executivesQuery.data;
   361→  const evaluatedIds = ps?.evaluatedExecutiveIds ?? [];
   362→  const evaluatedSet = new Set(evaluatedIds);
   363→
   364→  // Required-but-unanswered env questions.
   365→  const envRequiredMissing = useMemo(() => {
   366→    if (!campaign.enableEnvironmentSurvey) return [];
   367→    if (ps?.environmentSubmitted) return [];
   368→    return campaign.environmentQuestions.filter(
   369→      (q) => q.isRequired && !envAnswers[q.id]
   370→    );
   371→  }, [campaign, envAnswers, ps]);
   372→
   373→  const envStepComplete =
   374→    !campaign.enableEnvironmentSurvey ||
   375→    ps?.environmentSubmitted ||
   376→    envRequiredMissing.length === 0;
   377→
   378→  // Future: required single_choice needs ≥1, multi_choice needs ≥1.
   379→  const futureRequiredMissing = useMemo(() => {
   380→    if (!campaign.enableFutureSurvey) return [];
   381→    if (ps?.futureSubmitted) return [];
   382→    return campaign.futureQuestions.filter((q) => {
   383→      if (!q.isRequired) return false;
   384→      const vals = futureAnswers[q.id] ?? [];
   385→      return vals.length === 0;
   386→    });
   387→  }, [campaign, futureAnswers, ps]);
   388→
   389→  const futureStepComplete =
   390→    !campaign.enableFutureSurvey ||
   391→    ps?.futureSubmitted ||
   392→    futureRequiredMissing.length === 0;
   393→
   394→  // Executive step: picker + questions answered (only required when there
   395→  // are executives to evaluate).
   396→  const totalAssigned = executives?.executives.length ?? 0;
   397→  const availableExecutives =
   398→    executives?.executives.filter(
   399→      (e) => !evaluatedSet.has(e.id)
   400→    ) ?? [];
   401→  const noExecsToEvaluate = totalAssigned === 0 || availableExecutives.length === 0;
   402→  const singleEvalDone =
   403→    !campaign.allowMultipleExecutiveEvaluations && evaluatedIds.length >= 1;
   404→
   405→  // For the current executive eval form: are all required leadership
   406→  // questions answered?
   407→  const execQuestions = execQuestionsQuery.data?.questions ?? [];
   408→  const execRequiredMissing = execQuestions.filter(
   409→    (q) => q.isRequired && !executiveAnswers[q.id]
   410→  );
   411→  const execFormReady =
   412→    !!currentExecutiveId &&
   413→    !execQuestionsQuery.isLoading &&
   414→    execQuestions.length > 0 &&
   415→    execRequiredMissing.length === 0;
   416→
   417→  const execStepReady =
   418→    noExecsToEvaluate || singleEvalDone || evaluatedIds.length > 0;
   419→
   420→  // ─── Step 4 "submit" gating ──────────────────────────────────────────
   421→  const allDone = Boolean(
   422→    (!campaign.enableEnvironmentSurvey || ps?.environmentSubmitted) &&
   423→      (!campaign.enableFutureSurvey || ps?.futureSubmitted) &&
   424→      (singleEvalDone ||
   425→        noExecsToEvaluate ||
   426→        (campaign.allowMultipleExecutiveEvaluations
   427→          ? evaluatedIds.length >= (campaign.minExecutives ?? 1)
   428→          : evaluatedIds.length >= 1))
   429→  );
   430→  const canFinish = allDone && confirmedFinal;
   431→
   432→  // ─── Progress % for the Progress bar ────────────────────────────────
   433→  const progressPct = ((step - 1) / 4) * 100;
   434→
   435→  // ─── Handlers ────────────────────────────────────────────────────────
   436→  const goNext = () => {
   437→    if (step < 4) setStep(step + 1);
   438→  };
   439→  const goBack = () => {
   440→    if (step > 1) setStep(step - 1);
   441→  };
   442→
   443→  const handleEnvSubmit = () => {
   444→    if (envRequiredMissing.length > 0) {
   445→      toast({
   446→        variant: "destructive",
   447→        title: "أسئلة مطلوبة",
   448→        description: MESSAGES.incompleteAnswers,
   449→      });
   450→      return;
   451→    }
   452→    envMutation.mutate();
   453→  };
   454→
   455→  const handleFutureSubmit = () => {
   456→    if (futureRequiredMissing.length > 0) {
   457→      toast({
   458→        variant: "destructive",
   459→        title: "أسئلة مطلوبة",
   460→        description: MESSAGES.incompleteAnswers,
   461→      });
   462→      return;
   463→    }
   464→    futureMutation.mutate();
   465→  };
   466→
   467→  const handleExecSubmit = () => {
   468→    if (!currentExecutiveId) return;
   469→    if (execRequiredMissing.length > 0) {
   470→      toast({
   471→        variant: "destructive",
   472→        title: "أسئلة مطلوبة",
   473→        description: MESSAGES.incompleteAnswers,
   474→      });
   475→      return;
   476→    }
   477→    evalMutation.mutate({
   478→      executiveId: currentExecutiveId,
   479→      answers: executiveAnswers,
   480→    });
   481→  };
   482→
   483→  const handleFinalSubmit = () => {
   484→    if (!allDone) {
   485→      toast({
   486→        variant: "destructive",
   487→        title: "غير مكتمل",
   488→        description: MESSAGES.incompleteAnswers,
   489→      });
   490→      return;
   491→    }
   492→    if (!confirmedFinal) {
   493→      toast({
   494→        variant: "destructive",
   495→        title: "التأكيد مطلوب",
   496→        description: "يرجى تأكيد الإقرار قبل الإرسال.",
   497→      });
   498→      return;
   499→    }
   500→    // No server call — all sections were already submitted in steps 1-3.
   501→    // Just clear in-progress state and navigate to the success screen.
   502→    resetWizard();
   503→    onFinish();
   504→  };
   505→
   506→  // ─── Loading / error UI helpers ──────────────────────────────────────
   507→  if (participationStatusQuery.isLoading || executivesQuery.isLoading) {
   508→    return <WizardSkeleton />;
   509→  }
   510→  if (participationStatusQuery.isError) {
   511→    const err =
   512→      participationStatusQuery.error instanceof ApiError
   513→        ? participationStatusQuery.error
   514→        : null;
   515→    return (
   516→      <WizardError
   517→        message={
   518→          err?.message ?? "تعذّر تحميل حالة المشاركة. حاول مرة أخرى."
   519→        }
   520→        status={err?.status}
   521→        onRetry={() => participationStatusQuery.refetch()}
   522→      />
   523→    );
   524→  }
   525→
   526→  // ─── Render ──────────────────────────────────────────────────────────
   527→  return (
   528→    <div className="space-y-5">
   529→      {/* Progress indicator (spec §8.2) */}
   530→      <div className="mx-auto max-w-3xl">
   531→        <div className="flex items-center justify-between gap-1 mb-2">
   532→          {STEPS.map((s, i) => {
   533→            const n = i + 1;
   534→            const isCurrent = step === n;
   535→            const isDone = step > n;
   536→            return (
   537→              <div
   538→                key={s.key}
   539→                className="flex items-center gap-2 flex-1 last:flex-none"
   540→              >
   541→                <div
   542→                  className={
   543→                    "flex items-center gap-2 rounded-full px-2.5 py-1 text-xs font-medium transition-colors " +
   544→                    (isCurrent
   545→                      ? "bg-primary text-primary-foreground"
   546→                      : isDone
   547→                      ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300"
   548→                      : "bg-muted text-muted-foreground")
   549→                  }
   550→                >
   551→                  <span className="grid place-items-center h-5 w-5 rounded-full bg-background/40 text-[10px] font-bold">
   552→                    {isDone ? (
   553→                      <CheckCircle2 className="h-3.5 w-3.5" />
   554→                    ) : (
   555→                      n
   556→                    )}
   557→                  </span>
   558→                  <span className="hidden sm:inline">{s.labelAr}</span>
   559→                </div>
   560→                {n < 4 ? (
   561→                  <div className="h-px flex-1 bg-border" />
   562→                ) : null}
   563→              </div>
   564→            );
   565→          })}
   566→        </div>
   567→        <Progress value={progressPct} className="h-1.5" />
   568→      </div>
   569→
   570→      <AnimatePresence mode="wait" initial={false}>
   571→        <motion.div
   572→          key={step}
   573→          initial={{ opacity: 0, x: 24 }}
   574→          animate={{ opacity: 1, x: 0 }}
   575→          exit={{ opacity: 0, x: -24 }}
   576→          transition={{ duration: 0.2 }}
   577→        >
